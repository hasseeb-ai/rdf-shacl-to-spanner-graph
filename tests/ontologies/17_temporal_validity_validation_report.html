4. GQL Query Cheatsheet & Engine Invariants

Pattern / Capability Spanner GQL Query Example
Active Prescriptions Traversal:
Find active medications prescribed for a specific patient within a time window
GRAPH TemporalPrescriptionGraph MATCH (p:Patient {PatientId: 'PAT-1001'})<-[:PRESCRIBED_FOR]-(rx:Prescription)-[:DISPENSES_MEDICATION]->(m:Medication) WHERE rx.IsActive = TRUE AND rx.ValidFrom <= CURRENT_TIMESTAMP() AND (rx.ValidTo IS NULL OR rx.ValidTo >= CURRENT_TIMESTAMP()) RETURN p.PatientName, m.MedicationName, rx.DosageMg, rx.ValidFrom, rx.ValidTo;
Multi-Hop Patient History:
Query all historical prescriptions and dispensed drugs for all patients
GRAPH TemporalPrescriptionGraph MATCH (rx:Prescription)-[:PRESCRIBED_FOR]->(p:Patient), (rx)-[:DISPENSES_MEDICATION]->(m:Medication) RETURN p.PatientId, p.PatientName, m.MedicationName, rx.DosageMg, rx.ValidFrom, rx.ValidTo ORDER BY rx.ValidFrom DESC;
Spanner Engine Invariant Verifications:
  • Label Signature Uniformity: All exposed property lists (Patient, Medication, Prescription) strictly match their underlying column definitions and data types.
  • Check Constraint Compliance: Temporal interval logic CK_Prescription_TemporalValidity CHECK (ValidTo IS NULL OR ValidTo >= ValidFrom) correctly enforces valid start/end bounds.
  • Referential Integrity: Embedded foreign keys (FK_Prescriptions_Patient and FK_Prescriptions_Medication) guarantee referential consistency before edge traversal.

-- =============================================================================
-- Google Cloud Spanner Physical Relational Schema
-- =============================================================================

CREATE TABLE Patients (
  PatientId STRING(64) NOT NULL,
  PatientName STRING(MAX)
) PRIMARY KEY (PatientId);

CREATE TABLE Medications (
  MedicationId STRING(64) NOT NULL,
  MedicationName STRING(MAX)
) PRIMARY KEY (MedicationId);

CREATE TABLE Prescriptions (
  PrescriptionId STRING(64) NOT NULL,
  PatientId STRING(64) NOT NULL,
  MedicationId STRING(64) NOT NULL,
  DosageMg NUMERIC NOT NULL,
  ValidFrom TIMESTAMP NOT NULL,
  ValidTo TIMESTAMP,
  IsActive BOOL,
  CONSTRAINT FK_Prescriptions_Patient FOREIGN KEY (PatientId) REFERENCES Patients (PatientId),
  CONSTRAINT FK_Prescriptions_Medication FOREIGN KEY (MedicationId) REFERENCES Medications (MedicationId),
  CONSTRAINT CK_Prescription_TemporalValidity CHECK (ValidTo IS NULL OR ValidTo >= ValidFrom)
) PRIMARY KEY (PrescriptionId);

-- =============================================================================
-- Google Cloud Spanner Property Graph Schema
-- =============================================================================

CREATE PROPERTY GRAPH TemporalPrescriptionGraph
  NODE TABLES (
    Patients
      LABEL Patient PROPERTIES (PatientId, PatientName),
    Medications
      LABEL Medication PROPERTIES (MedicationId, MedicationName),
    Prescriptions
      LABEL Prescription PROPERTIES (PrescriptionId, DosageMg, ValidFrom, ValidTo, IsActive)
  )
  EDGE TABLES (
    Prescriptions AS PrescribedForEdges
      SOURCE KEY (PrescriptionId) REFERENCES Prescriptions (PrescriptionId)
      DESTINATION KEY (PatientId) REFERENCES Patients (PatientId)
      LABEL PRESCRIBED_FOR,
    Prescriptions AS DispensesMedicationEdges
      SOURCE KEY (PrescriptionId) REFERENCES Prescriptions (PrescriptionId)
      DESTINATION KEY (MedicationId) REFERENCES Medications (MedicationId)
      LABEL DISPENSES_MEDICATION
  );
