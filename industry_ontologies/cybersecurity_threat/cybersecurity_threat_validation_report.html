CREATE VIEW CriticalSeverityAlerts SQL SECURITY INVOKER AS
SELECT
  sa.SecurityAlertId,
  sa.SeverityScore,
  sa.IsCriticalSeverityAlert
FROM SecurityAlerts sa
WHERE sa.SeverityScore >= 9;
